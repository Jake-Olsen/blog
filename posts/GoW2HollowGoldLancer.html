Unlock Gold Lancer and Unlock Gold Hammburst by editing GearGame.ini

Add this section after line 857 to get the Gold Hammerburst and Gold Lancer.
Then you can adjust it on loadout / default loadout.

[GoldLancer GearGameWeaponSummary]
ClassPathName=GearGameContent.GearWeap_AssaultRifle_Golden
WeaponIcon=(IconPathName="UI_Portraits.Party_GOLD_Lancer",Coordinates=(U=0,V=0,UL=256,VL=128))
ProfileId=eGEARWEAP_GoldLancer
bIsDefaultWeapon=true
WeaponId=17

[GoldHammer GearGameWeaponSummary]
ClassPathName=GearGameContent.GearWeap_LocustAssaultRifle_Golden
WeaponIcon=(IconPathName="UI_Portraits.Party_GOLD_Hammerburst",Coordinates=(U=0,V=0,UL=256,VL=128))
ProfileId=eGEARWEAP_GoldHammer
bIsDefaultWeapon=true
WeaponId=18
